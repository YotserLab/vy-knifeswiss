> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Módulo "Cast to TV" — descoberta de dispositivos (DLNA/Roku/Chromecast/DIAL) + cast de mídia local (vídeo/áudio/foto) com mini-controle remoto, integrado ao FileManager e MediaPlayer.

**Architecture:** Pacote `com.example.cast` com 4 camadas (discovery/control/media/ui). Descoberta via SSDP M-SEARCH (DLNA/DIAL/Roku) + mDNS `_googlecast._tcp` (Chromecast). Mídia servida por `InternalHttpFileServer` estendido (mime + HTTP Range + TTL). Controle: SOAP AVTransport manual (DLNA), ECP manual (Roku), Google Cast SDK (Chromecast). Todo HTTP LAN via socket TCP bruto.

**Tech Stack:** Kotlin, Jetpack Compose M3, coroutines/Flow, NsdManager, XmlPullParser, play-services-cast-framework + androidx.mediarouter (novas deps), JUnit + Robolectric (testes).

```kotlin
package com.example.cast.discovery

enum class CastDeviceType(val priority: Int) {
    DIAL(0), ROKU(1), DLNA(2), CHROMECAST(3)
}

data class CastDevice(
    val id: String,
    val name: String,
    val type: CastDeviceType,
    val ip: String,
    val port: Int = 0,
    val manufacturer: String = "",
    val modelName: String = "",
    val location: String = "",
    val avTransportControlUrl: String = "",
    val renderingControlUrl: String = ""
) {
    val canCastLocalMedia: Boolean
        get() = when (type) {
            CastDeviceType.DLNA -> avTransportControlUrl.isNotEmpty()
            CastDeviceType.ROKU, CastDeviceType.CHROMECAST -> true
            CastDeviceType.DIAL -> false
        }
}

object CastDeviceMerger {
    fun merge(existing: List<CastDevice>, incoming: List<CastDevice>): List<CastDevice> {
        val byIp = (existing + incoming).groupBy { it.ip }
        return byIp.map { (_, group) ->
            group.filter { it.canCastLocalMedia }.maxByOrNull { it.type.priority }
                ?: group.maxByOrNull { it.type.priority }!!
        }.sortedBy { it.name.lowercase() }
    }
}
```

```kotlin
package com.example.cast.control

import java.io.OutputStream
import java.net.InetSocketAddress
import java.net.Socket
import java.net.URI

object LanHttpClient {
    data class Response(val code: Int, val headers: Map<String, String>, val body: String)

    fun get(url: String, timeoutMs: Int = 5000): Response? =
        request("GET", url, null, "text/plain", emptyMap(), timeoutMs)

    fun post(url: String, body: String, contentType: String,
             extraHeaders: Map<String, String> = emptyMap(), timeoutMs: Int = 8000): Response? =
        request("POST", url, body, contentType, extraHeaders, timeoutMs)

    private fun request(method: String, url: String, body: String?, contentType: String,
                        extraHeaders: Map<String, String>, timeoutMs: Int): Response? {
        return try {
            val uri = URI(url)
            val host = uri.host ?: return null
            val port = if (uri.port > 0) uri.port else 80
            val path = (uri.rawPath ?: "/") + (uri.rawQuery?.let { "?$it" } ?: "")
            Socket().use { socket ->
                socket.connect(InetSocketAddress(host, port), timeoutMs)
                socket.soTimeout = timeoutMs
                val out: OutputStream = socket.getOutputStream()
                val sb = StringBuilder("$method $path HTTP/1.1\r\nHost: $host:$port\r\nConnection: close\r\n")
                if (body != null) {
                    val bytes = body.toByteArray(Charsets.UTF_8)
                    sb.append("Content-Type: $contentType; charset=\"utf-8\"\r\nContent-Length: ${bytes.size}\r\n")
                }
                extraHeaders.forEach { (k, v) -> sb.append("$k: $v\r\n") }
                sb.append("\r\n")
                out.write(sb.toString().toByteArray(Charsets.ISO_8859_1))
                if (body != null) out.write(body.toByteArray(Charsets.UTF_8))
                out.flush()

                val input = socket.getInputStream()
                val raw = input.readBytes()
                parseResponse(String(raw, Charsets.ISO_8859_1))
            }
        } catch (_: Exception) { null }
    }

    internal fun parseResponse(raw: String): Response? {
        val sep = raw.indexOf("\r\n\r\n")
        if (sep < 0) return null
        val head = raw.substring(0, sep)
        val body = raw.substring(sep + 4)
        val lines = head.split("\r\n")
        val code = lines.firstOrNull()?.split(" ")?.getOrNull(1)?.toIntOrNull() ?: return null
        val headers = lines.drop(1).mapNotNull {
            val i = it.indexOf(':'); if (i < 0) null else it.substring(0, i).lowercase() to it.substring(i + 1).trim()
        }.toMap()
        return Response(code, headers, body)
    }
}
```

```kotlin
package com.example.cast.discovery

import java.net.URI

data class SsdpResponse(
    val location: String, val st: String, val usn: String, val server: String
) {
    val host: String get() = runCatching { URI(location).host }.getOrNull() ?: ""
    val port: Int get() = runCatching { URI(location).port.takeIf { it > 0 } ?: 80 }.getOrDefault(80)
    val udnUuid: String get() = usn.substringBefore("::").removePrefix("uuid:")
}

object SsdpParser {
    fun parse(raw: String): SsdpResponse? {
        val headers = raw.lines().mapNotNull {
            val i = it.indexOf(':'); if (i < 0) null else it.substring(0, i).trim().lowercase() to it.substring(i + 1).trim()
        }.toMap()
        val location = headers["location"] ?: return null
        return SsdpResponse(
            location = location,
            st = headers["st"] ?: headers["nt"] ?: "",
            usn = headers["usn"] ?: "",
            server = headers["server"] ?: ""
        )
    }
}
```

```kotlin
class SsdpDiscovery(private val context: Context) {
    suspend fun scan(windowMs: Long = 4000): List<SsdpResponse> = withContext(Dispatchers.IO) {
        val wifi = context.getSystemService(Context.WIFI_SERVICE) as WifiManager
        val lock = wifi.createMulticastLock("cast_ssdp").apply { setReferenceCounted(true); acquire() }
        try {
            val socket = DatagramSocket().apply { soTimeout = 500 }
            val group = InetAddress.getByName("239.255.255.250")
            val targets = listOf(
                "ssdp:all",
                "urn:schemas-upnp-org:device:MediaRenderer:1",
                "urn:dial-multiscreen-org:service:dial:1",
                "roku:ecp"
            )
            repeat(2) {
                targets.forEach { st ->
                    val msg = "M-SEARCH * HTTP/1.1\r\nHOST: 239.255.255.250:1900\r\n" +
                        "MAN: \"ssdp:discover\"\r\nMX: 3\r\nST: $st\r\n\r\n"
                    runCatching { socket.send(DatagramPacket(msg.toByteArray(), msg.length, group, 1900)) }
                }
                delay(300)
            }
            val results = LinkedHashMap<String, SsdpResponse>()
            val deadline = System.currentTimeMillis() + windowMs
            val buf = ByteArray(8192)
            while (System.currentTimeMillis() < deadline) {
                try {
                    val pkt = DatagramPacket(buf, buf.size)
                    socket.receive(pkt)
                    val raw = String(pkt.data, 0, pkt.length, Charsets.ISO_8859_1)
                    SsdpParser.parse(raw)?.let { results.putIfAbsent(it.location, it) }
                } catch (_: SocketTimeoutException) {}
            }
            socket.close()
            results.values.toList()
        } finally { runCatching { lock.release() } }
    }
```

```kotlin
class CastDiscovery(private val context: Context) {
    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.Main.immediate)
    private val _devices = MutableStateFlow<List<CastDevice>>(emptyList())
    val devices: StateFlow<List<CastDevice>> = _devices
    val scanning = MutableStateFlow(false)
    private val mdns = MdnsCastDiscovery(context)

    fun startScan() {
        if (scanning.value) return
        scanning.value = true
        mdns.start { dev -> mergeIn(listOf(dev)) }
        scope.launch {
            try {
                val responses = SsdpDiscovery(context).scan()
                val found = responses.mapNotNull { resp -> describe(resp) }
                mergeIn(found)
            } finally { scanning.value = false }
        }
        scope.launch {
            delay(6000); scanning.value = false; mdns.stop()
        }
    }

    private suspend fun describe(resp: SsdpResponse): CastDevice? {
        if (resp.st.contains("roku:ecp")) return RokuProbe.probe(resp.host)
        val xml = LanHttpClient.get(resp.location)?.takeIf { it.code == 200 }?.body ?: return null
        val d = DeviceDescParser.parse(xml, "${resp.location.substringBefore("/desc")}") ?: return null
        val type = when {
            d.avTransportControlUrl.isNotEmpty() -> CastDeviceType.DLNA
            resp.st.contains("dial") || d.manufacturer.contains("Google", true) -> CastDeviceType.DIAL
            else -> CastDeviceType.DIAL
        }
        return CastDevice(d.udn.ifEmpty { resp.location }, d.friendlyName.ifEmpty { resp.host },
            type, resp.host, resp.port, d.manufacturer, d.modelName,
            resp.location, d.avTransportControlUrl, d.renderingControlUrl)
    }

    fun addManualDevice(ip: String) { scope.launch { RokuProbe.probe(ip)?.let { mergeIn(listOf(it)) } } }
    private fun mergeIn(new: List<CastDevice>) { _devices.value = CastDeviceMerger.merge(_devices.value, new) }
    fun clear() { _devices.value = emptyList() }
}
```

```kotlin
object UpnpSoap {
    const val AVT = "urn:schemas-upnp-org:service:AVTransport:1"
    const val RCS = "urn:schemas-upnp-org:service:RenderingControl:1"

    fun envelope(action: String, serviceType: String, args: Map<String, String>): String {
        val inner = args.entries.joinToString("") { (k, v) -> "<$k>$v</$k>" }
        return """<?xml version="1.0" encoding="utf-8"?>
<s:Envelope xmlns:s="http://schemas.xmlsoap.org/soap/envelope/" s:encodingStyle="http://schemas.xmlsoap.org/soap/encoding/">
<s:Body><u:$action xmlns:u="$serviceType">$inner</u:$action></s:Body></s:Envelope>"""
    }

    fun didlLite(title: String, url: String, mime: String, kind: CastMedia.Kind, durationMs: Long): String {
        val cls = when (kind) {
            CastMedia.Kind.VIDEO -> "object.item.videoItem"
            CastMedia.Kind.AUDIO -> "object.item.audioItem.musicTrack"
            CastMedia.Kind.IMAGE -> "object.item.imageItem.photo"
        }
        val dur = if (durationMs > 0) " duration=\"${msToHms(durationMs)}\"" else ""
        return """<DIDL-Lite xmlns="urn:schemas-upnp-org:metadata-1-0/DIDL-Lite/" xmlns:dc="http://purl.org/dc/elements/1.1/" xmlns:upnp="urn:schemas-upnp-org:metadata-1-0/upnp/"><item id="0" parentID="-1" restricted="1"><dc:title>${escape(title)}</dc:title><upnp:class>$cls</upnp:class><res protocolInfo="http-get:*:$mime:*"$dur>${escape(url)}</res></item></DIDL-Lite>"""
    }

    fun escape(s: String) = s.replace("&","&amp;").replace("<","&lt;").replace(">","&gt;")
        .replace("\"","&quot;").replace("'","&apos;")

    fun parseHmsToMs(hms: String): Long { /* split ":", 2 ou 3 partes → ms */ }
    fun msToHms(ms: Long): String { /* H:MM:SS */ }
    fun parsePositionInfo(body: String): PositionInfo { /* RelTime, TrackDuration, TransportState via substringAfter/tag */ }
    fun parseVolume(body: String): Int { /* <CurrentVolume> */ }
}
data class PositionInfo(val relTimeMs: Long, val durationMs: Long, val state: String)
```
